\clist_new:N \g_cc_costi_clist

\fp_compare:nNnF { \Durata } = { \g_cc_n_int }
  { \cc_errore:n { PianoCosti~contiene~\int_use:N \g_cc_n_int \c_space_tl valori~ma~Durata~vale~\Durata.~Devono~coincidere. } }
\fp_new:N \g_cc_cum_fp
\cc_set:nn { bcws0 } { 0 }
\int_step_inline:nn { \g_cc_n_int }
  {
    \fp_compare:nNnT { \clist_item:Nn \g_cc_costi_clist {#1} } < { 0 }
      { \cc_errore:n { il~costo~del~periodo~#1~e'~negativo. } }
    \fp_gadd:Nn \g_cc_cum_fp { \clist_item:Nn \g_cc_costi_clist {#1} }
    \cc_set:nn { bcws#1 } { \fp_use:N \g_cc_cum_fp }
    \cc_set:nn { costo#1 } { \fp_eval:n { \clist_item:Nn \g_cc_costi_clist {#1} } }

% --- 2. Punti di BCWP e ACWP (coppie periodo/valore cumulato) ---------
\seq_new:N \l_cc_coppia_seq
\cs_new_protected:Npn \cc_punti:nn #1 #2
  {
    % #1 = bcwp o acwp, #2 = lista "periodo/valore, ..."
    \seq_new:c { g_cc_#1x_seq } \seq_new:c { g_cc_#1y_seq }
    \clist_set:Nx \l_tmpa_clist {#2}
    \clist_map_inline:Nn \l_tmpa_clist
      {
        \seq_set_split:Nnn \l_cc_coppia_seq { / } {##1}
        \seq_gput_right:cx { g_cc_#1x_seq } { \fp_eval:n { \seq_item:Nn \l_cc_coppia_seq { 1 } } }
        \seq_gput_right:cx { g_cc_#1y_seq } { \fp_eval:n { \seq_item:Nn \l_cc_coppia_seq { 2 } } }
      }
    \int_compare:nNnT { \seq_count:c { g_cc_#1x_seq } } = { 0 }
      { \cc_errore:n { la~lista~dei~punti~di~#1~e'~vuota. } }
    \int_step_inline:nnn { 2 } { \seq_count:c { g_cc_#1x_seq } }
      {
        \fp_compare:nNnF { \seq_item:cn { g_cc_#1x_seq } { ##1 } } > { \seq_item:cn { g_cc_#1x_seq } { ##1 - 1 } }
          { \cc_errore:n { i~punti~di~#1~devono~essere~in~ordine~di~periodo~crescente. } }
        \fp_compare:nNnT { \seq_item:cn { g_cc_#1y_seq } { ##1 } } < { \seq_item:cn { g_cc_#1y_seq } { ##1 - 1 } }
          { \cc_errore:n { i~valori~di~#1~sono~cumulati~e~non~possono~diminuire. } }
      }
    \cc_set:nn { #1ultimoT } { \seq_item:cn { g_cc_#1x_seq } { -1 } }
    \cc_set:nn { #1ultimoV } { \seq_item:cn { g_cc_#1y_seq } { -1 } }
    \cc_set:nn { #1penT } { \int_compare:nNnTF { \seq_count:c { g_cc_#1x_seq } } > { 1 } { \seq_item:cn { g_cc_#1x_seq } { -2 } } { 0 } }
    \cc_set:nn { #1penV } { \int_compare:nNnTF { \seq_count:c { g_cc_#1x_seq } } > { 1 } { \seq_item:cn { g_cc_#1y_seq } { -2 } } { 0 } }
  }
\cc_punti:nn { bcwp } { \PuntiBCWP }
\cc_punti:nn { acwp } { \PuntiACWP }
% valore di una serie in un periodo (deve esserci)
\cs_new_protected:Npn \cc_cerca:nnn #1 #2 #3
  {
    % #1 = serie, #2 = periodo, #3 = nome del risultato
    \cc_set:nn {#3} { nd }
    \int_step_inline:nn { \seq_count:c { g_cc_#1x_seq } }
      {
        \fp_compare:nNnT { \seq_item:cn { g_cc_#1x_seq } { ##1 } } = { #2 }
          { \cc_set:nn {#3} { \seq_item:cn { g_cc_#1y_seq } { ##1 } } }
      }
    \str_if_eq:eeT { \cc_v:n {#3} } { nd }
      { \cc_errore:n { al~periodo~#2~(rilevamento)~manca~il~valore~di~#1~nella~lista~Punti. } }
  }

% --- 3. Indicatori di un rilevamento (formule del libro, pag. 199) ---
\cs_new_protected:Npn \cc_rilevamento:nn #1 #2
  {
    % #1 = A o B, #2 = periodo
    \bool_if:nF { \fp_compare_p:n { #2 = round(#2) } && \fp_compare_p:n { #2 >= 1 } && \fp_compare_p:n { #2 <= \g_cc_n_int } }
      { \cc_errore:n { il~periodo~del~rilevamento~#1~deve~essere~un~intero~tra~1~e~\int_use:N \g_cc_n_int. } }
    \cc_set:nn { t#1 } { \int_eval:n { \fp_eval:n { round(#2) } } }
    \cc_set:nn { bcws#1 } { \cc_bcws:n { \cc_v:n { t#1 } } }
    \cc_cerca:nnn { bcwp } { \cc_v:n { t#1 } } { bcwp#1 }
    \cc_cerca:nnn { acwp } { \cc_v:n { t#1 } } { acwp#1 }
    \cc_set:nn { cv#1 } { \fp_eval:n { \cc_v:n { bcwp#1 } - \cc_v:n { acwp#1 } } }   % CV = BCWP - ACWP
    \cc_set:nn { sv#1 } { \fp_eval:n { \cc_v:n { bcwp#1 } - \cc_v:n { bcws#1 } } }   % SV = BCWP - BCWS
    \fp_compare:nNnTF { \cc_v:n { bcws#1 } } > { 0 }
      {
        \cc_set:nn { pcv#1 } { \fp_eval:n { \cc_v:n { cv#1 } / \cc_v:n { bcws#1 } * 100 } }  % %CV = CV / BCWS
        \cc_set:nn { psv#1 } { \fp_eval:n { \cc_v:n { sv#1 } / \cc_v:n { bcws#1 } * 100 } }  % %SV = SV / BCWS
        \cc_set:nn { spi#1 } { \fp_eval:n { \cc_v:n { bcwp#1 } / \cc_v:n { bcws#1 } } }      % SPI (assunzione)
        \cc_set:nn { haperc#1 } { 1 }
      }
      { \cc_set:nn { haperc#1 } { 0 } \cc_set:nn { spi#1 } { nd } }
    \fp_compare:nNnTF { \cc_v:n { acwp#1 } } > { 0 }
      { \cc_set:nn { cpi#1 } { \fp_eval:n { \cc_v:n { bcwp#1 } / \cc_v:n { acwp#1 } } } }    % CPI (assunzione)
      { \cc_set:nn { cpi#1 } { nd } }
  }
\cc_rilevamento:nn { A } { \TempoA }
\tl_if_blank:oTF { \TempoB }
  { \bool_gset_false:N \g_cc_due_bool \cc_set:nn { U } { A } }
  {
    \bool_gset_true:N \g_cc_due_bool \cc_set:nn { U } { B }
    \cc_rilevamento:nn { B } { \TempoB }
    \int_compare:nNnF { \cc_v:n { tA } } < { \cc_v:n { tB } }
      { \cc_errore:n { TempoB~deve~essere~maggiore~di~TempoA. } }
  }
\cs_new:Npn \cc_u:n #1 { \cc_v:n { #1 \cc_v:n { U } } }    % valore all'ultimo rilevamento

% --- 4. Previsione: auto (EAC = BAC / CPI, fine = Durata / SPI) o manuale
\cc_set:nn { motivo } { }
\str_case:enF { \Previsione }
  {
    { auto }
      {
        \bool_if:nTF { \str_if_eq_p:ee { \cc_u:n { cpi } } { nd } || \str_if_eq_p:ee { \cc_u:n { spi } } { nd } }
          { \cc_set:nn { motivo } { Previsione~non~calcolabile:~servono~BCWP,~ACWP~e~BCWS~maggiori~di~zero~all'ultimo~rilevamento. } }
          {
            \fp_compare:nNnTF { \cc_u:n { cpi } * \cc_u:n { spi } } > { 0 }
              {
                \cc_set:nn { eac } { \fp_eval:n { \BAC / \cc_u:n { cpi } } }
                \cc_set:nn { fine } { \fp_eval:n { \g_cc_n_int / \cc_u:n { spi } } }
              }
              { \cc_set:nn { motivo } { Previsione~non~calcolabile:~servono~BCWP,~ACWP~e~BCWS~maggiori~di~zero~all'ultimo~rilevamento. } }
          }
      }
    { manuale }
      {
        \cc_set:nn { eac } { \fp_eval:n { \EACManuale } }
        \cc_set:nn { fine } { \fp_eval:n { \FineManuale } }
      }
    { no } { \cc_set:nn { motivo } { - } }
  }
  { \cc_errore:n { Previsione~deve~valere~auto,~manuale~oppure~no. } }
\tl_if_blank:eT { \cc_v:n { motivo } }
  {
    \fp_compare:nNnTF { \cc_v:n { fine } } < { \cc_v:n { acwpultimoT } + 0.000000001 }
      { \cc_set:nn { motivo } { La~fine~stimata~cade~prima~dell'ultimo~costo~rilevato:~la~curva~di~previsione~non~si~disegna. } }
      {
        \fp_compare:nNnTF { \cc_v:n { eac } } < { \cc_v:n { acwpultimoV } - 0.000000001 }
          { \cc_set:nn { motivo } { Il~costo~finale~stimato~e'~minore~di~quanto~gia'~speso:~la~curva~di~previsione~non~si~disegna. } }
          {
            \bool_if:nTF { \str_if_eq_p:ee { \Previsione } { auto } && \fp_compare_p:n { \cc_v:n { fine } > 3 * \g_cc_n_int } }
              { \cc_set:nn { motivo } { La~fine~stimata~supera~il~triplo~della~durata~pianificata:~la~curva~di~previsione~non~si~disegna. } }
              { \bool_gset_true:N \g_cc_prev_bool }
          }
      }
  }
\str_if_eq:eeT { \cc_v:n { motivo } } { - } { \cc_set:nn { motivo } { } }

% --- 5. Assi e vista (zoom): stesso algoritmo del sito ---------------
\cs_new_protected:Npn \cc_passo_bello:nn #1 #2
  {
    % #1 = nome del risultato, #2 = valore. Passo 1, 2, 2,5, 5 o 10 x 10^e
    \fp_set:Nn \l_cc_a_fp {#2}
    \fp_compare:nNnTF \l_cc_a_fp > 0

        \fp_set:Nn \l_cc_b_fp { 10 ^ floor( ln(\l_cc_a_fp) / ln(10) ) }

        \cc_set:nn {#1} { \fp_eval:n { ( \l_cc_c_fp <= 1 + 1e-9 ? 1 : ( \l_cc_c_fp <= 2 + 1e-9 ? 2 : ( \l_cc_c_fp <= 2.5 + 1e-9 ? 2.5 : ( \l_cc_c_fp <= 5 + 1e-9 ? 5 : 10 ) ) ) ) * \l_cc_b_fp } }
      }
      { \cc_set:nn {#1} { 1 } }
  }

% --- 6. Curve: Bézier monotone (Fritsch-Butland), come nel sito ------
\seq_new:N \l_cc_x_seq
\seq_new:N \l_cc_y_seq
\int_new:N \l_cc_k_int
\tl_new:N \l_cc_path_tl
\cs_new:Npn \cc_x:n #1 { \seq_item:Nn \l_cc_x_seq { #1 } }
\cs_new:Npn \cc_y:n #1 { \seq_item:Nn \l_cc_y_seq { #1 } }
\cs_new:Npn \cc_h:n #1 { \tl_use:c { l_cc_h #1 _tl } }
\cs_new:Npn \cc_d:n #1 { \tl_use:c { l_cc_d #1 _tl } }
\cs_new:Npn \cc_m:n #1 { \tl_use:c { l_cc_m #1 _tl } }
\cs_new_protected:Npn \cc_bezier:n #1
  {
    % usa i nodi in \l_cc_x_seq e \l_cc_y_seq e salva il percorso come g_cc_path#1_tl
    \int_set:Nn \l_cc_k_int { \seq_count:N \l_cc_x_seq - 1 }
    \int_step_inline:nn { \l_cc_k_int }
      {
        \tl_set:cx { l_cc_h##1_tl } { \fp_eval:n { \cc_x:n { ##1 + 1 } - \cc_x:n { ##1 } } }
        \tl_set:cx { l_cc_d##1_tl } { \fp_eval:n { ( \cc_y:n { ##1 + 1 } - \cc_y:n { ##1 } ) / \cc_h:n { ##1 } } }
      }
    \tl_set:cn { l_cc_m1_tl } { 0 }
    \tl_set:cx { l_cc_m \int_eval:n { \l_cc_k_int + 1 } _tl } { \cc_d:n { \int_use:N \l_cc_k_int } }
    \int_step_inline:nnn { 2 } { \l_cc_k_int }
      {
        \fp_compare:nNnTF { \cc_d:n { \int_eval:n { ##1 - 1 } } * \cc_d:n { ##1 } } > { 0 }
          {
            \tl_set:cx { l_cc_m##1_tl } { \fp_eval:n {
              3 * ( \cc_h:n { \int_eval:n { ##1 - 1 } } + \cc_h:n { ##1 } ) /
              ( ( 2 * \cc_h:n { ##1 } + \cc_h:n { \int_eval:n { ##1 - 1 } } ) / \cc_d:n { \int_eval:n { ##1 - 1 } }
              + ( \cc_h:n { ##1 } + 2 * \cc_h:n { \int_eval:n { ##1 - 1 } } ) / \cc_d:n { ##1 } ) } }
          }
          { \tl_set:cn { l_cc_m##1_tl } { 0 } }
      }
    \tl_set:Nx \l_cc_path_tl { ( \cc_x:n { 1 } , \cc_y:n { 1 } ) }
    \int_step_inline:nn { \l_cc_k_int }
      {
        \tl_put_right:Nx \l_cc_path_tl
          {
            \c_space_tl .. \c_space_tl controls \c_space_tl
            ( \fp_eval:n { \cc_x:n { ##1 } + \cc_h:n { ##1 } / 3 } , \fp_eval:n { \cc_y:n { ##1 } + \cc_m:n { ##1 } * \cc_h:n { ##1 } / 3 } )
            \c_space_tl and \c_space_tl
            ( \fp_eval:n { \cc_x:n { ##1 + 1 } - \cc_h:n { ##1 } / 3 } , \fp_eval:n { \cc_y:n { ##1 + 1 } - \cc_m:n { \int_eval:n { ##1 + 1 } } * \cc_h:n { ##1 } / 3 } )
            \c_space_tl .. \c_space_tl
            ( \cc_x:n { ##1 + 1 } , \cc_y:n { ##1 + 1 } )
          }
      }
    \tl_gset_eq:cN { g_cc_path#1_tl } \l_cc_path_tl
  }
% BCWS: un nodo per periodo
\seq_clear:N \l_cc_x_seq \seq_clear:N \l_cc_y_seq
\int_step_inline:nnn { 0 } { \g_cc_n_int }
  { \seq_put_right:Nn \l_cc_x_seq { #1 } \seq_put_right:Nx \l_cc_y_seq { \cc_bcws:n { #1 } } }
\cc_bezier:n { BCWS }
% BCWP e ACWP: l'origine e tutti i punti inseriti
\clist_map_inline:nn { bcwp, acwp }
  {
    \seq_clear:N \l_cc_x_seq \seq_clear:N \l_cc_y_seq
    \seq_put_right:Nn \l_cc_x_seq { 0 } \seq_put_right:Nn \l_cc_y_seq { 0 }
    \int_step_inline:nn { \seq_count:c { g_cc_#1x_seq } }
      {
        \fp_compare:nNnT { \seq_item:cn { g_cc_#1x_seq } { ##1 } } > { 0 }
          {
            \seq_put_right:Nx \l_cc_x_seq { \seq_item:cn { g_cc_#1x_seq } { ##1 } }
            \seq_put_right:Nx \l_cc_y_seq { \seq_item:cn { g_cc_#1y_seq } { ##1 } }
          }
      }
    \cc_bezier:n { #1 }
  }
% Previsione: dall'ultimo costo rilevato a (fine, EAC), tangente finale 0
\bool_if:NT \g_cc_prev_bool
  {

    \fp_set:Nn \l_cc_b_fp { max( 0 , min( ( \cc_v:n { acwpultimoV } - \cc_v:n { acwppenV } ) / ( \cc_v:n { acwpultimoT } - \cc_v:n { acwppenT } ) ,
                                         3 * ( \cc_v:n { eac } - \cc_v:n { acwpultimoV } ) / \l_cc_a_fp ) ) }     % m
    \cc_set:nn { pathPrev }
      {
        ( \cc_v:n { acwpultimoT } , \cc_v:n { acwpultimoV } ) \c_space_tl .. \c_space_tl controls \c_space_tl
        ( \fp_eval:n { \cc_v:n { acwpultimoT } + \l_cc_a_fp / 3 } , \fp_eval:n { \cc_v:n { acwpultimoV } + \l_cc_b_fp * \l_cc_a_fp / 3 } )
        \c_space_tl and \c_space_tl
        ( \fp_eval:n { \cc_v:n { fine } - \l_cc_a_fp / 3 } , \cc_v:n { eac } )

  }

\section*{\large Legenda dei parametri}
\begin{small}
\begin{description}\setlength{\itemsep}{1pt}
  \item[BCWS] \emph{Budgeted Cost of Work Scheduled} -- pianificato a budget (baseline): quanto si doveva spendere fino a quel momento secondo il piano. Il valore finale è il budget totale (BAC).
  \item[BCWP] \emph{Budgeted Cost of Work Performed} -- lavoro svolto a costi di budget (\emph{earned value}): quanto vale, ai prezzi del piano, il lavoro davvero fatto.
  \item[ACWP] \emph{Actual Cost of Work Performed} -- costo effettivamente sostenuto per il lavoro fatto.
  \item[CV] \emph{Cost Variance} $=$ BCWP $-$ ACWP. Negativa: si è speso più del valore del lavoro svolto (sforamento dei costi).
  \item[SV] \emph{Schedule Variance} $=$ BCWP $-$ BCWS. Negativa: si è fatto meno lavoro del previsto (ritardo), espresso in denaro.
  \item[\%CV, \%SV] \%CV(\textit{t}) $=$ CV(\textit{t}) / BCWS(\textit{t}) e \%SV(\textit{t}) $=$ SV(\textit{t}) / BCWS(\textit{t}): andamento percentuale nel tempo (formule del libro).
  \item[CPI, SPI, EAC] \emph{Assunzioni, non nel libro}: CPI $=$ BCWP/ACWP, SPI $=$ BCWP/BCWS, costo finale stimato EAC $=$ BAC/CPI, fine stimata $=$ durata/SPI (ultimo rilevamento), salvo previsione inserita a mano.
\end{description}
\end{small}

\section*{\large Nota sulle formule}
{\footnotesize
Formule del libro (pagg.~198--199): CV $=$ BCWP $-$ ACWP, SV $=$ BCWP $-$ BCWS, \%CV $=$ CV/BCWS, \%SV $=$ SV/BCWS. In letteratura la \%CV si calcola spesso su BCWP invece che su BCWS: qui si segue il libro.
Le curve passano esattamente per i valori inseriti con un'interpolazione cubica monotona (Fritsch--Butland), quindi non scendono mai.
Semaforo: \emph{in linea} se CV e SV non sono negativi; \emph{critico} se \%CV o \%SV sono sotto il $-10\%$ (soglia ipotizzata); altrimenti \emph{attenzione}.
